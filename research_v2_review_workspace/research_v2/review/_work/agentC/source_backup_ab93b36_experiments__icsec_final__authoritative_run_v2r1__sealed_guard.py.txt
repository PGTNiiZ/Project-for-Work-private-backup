1: """Process-wide sealed-label value guard and runtime access audit for v2r1.
2: 
3: The scientific invariant is defined precisely as:
4: 
5:     SEALED_TEST LABEL VALUES MATERIALIZED BEFORE FINAL_EVALUATION == 0
6: 
7: "Materialized" means the ground-truth ``actual`` column of a registered
8: sealed-label artifact is loaded into memory. Hashing bytes, reading parquet
9: footers/schema/row counts, or loading ONLY the safe ``pair_key`` column are
10: explicitly allowed pre-final and are audited as non-materializing events.
11: 
12: The guard patches ``pandas.read_parquet`` and ``pyarrow.parquet.read_table``
13: for the whole process, so every code path used by the scientific CLI is
14: covered — not only ``LazySealedLabelStore``.
15: """

24: import pyarrow.parquet as pq

30:     """Raised when sealed-test label values would be materialized pre-final."""

34: class SealedLabelGuard:
35:     """Singleton-style runtime guard over registered sealed-label artifacts."""

37:     sealed_paths: set[str] = field(default_factory=set)

44:     def register(self, path: str | Path) -> None:

50:                             "order": len(self.events), "timestamp": time.time()})

53:         try:

61:             "event": "sealed_artifact_access", "phase": self.phase,
62:             "resource": resource, "operation": operation,
63:             "requested_columns": list(columns) if columns is not None else None,
64:             "label_column_materialized": materialized, "allowed": allowed,
65:             "order": len(self.events), "timestamp": time.time(),

68:     def _check(self, path: object, columns: Any, operation: str):

70:             return
71:         materializes = columns is None or LABEL_COLUMN in set(columns)
72:         allowed = (not materializes) or self.phase == "FINAL_EVALUATION"
73:         self._audit(resource=str(path), operation=operation, columns=columns,
74:                     materialized=materializes and allowed, allowed=allowed)
75:         if not allowed:
76:             raise SealedLabelViolation(
77:                 f"sealed-test label values may not be materialized during "
78:                 f"phase {self.phase!r} (path={path})")

84:         self._original_pq_read = pq.read_table
85:         guard = self

87:         def guarded_pd_read(path, *args, **kwargs):
88:             guard._check(path, kwargs.get("columns"), "pandas.read_parquet")
89:             return guard._original_pd_read(path, *args, **kwargs)
90: 
91:         def guarded_pq_read(source, *args, **kwargs):

102:             pq.read_table = self._original_pq_read

113:         return sum(1 for e in self.events
