80: def bloom_packed(s: str, L: int, k: int, salt: str) -> tuple[np.ndarray, int]:
81:     """คืน packed bits (uint8, ceil(L/8) bytes) + popcount ของ string s"""
82:     bits = np.zeros(L, dtype=bool)
83:     for g in bigrams(s):
84:         for j in range(k):
85:             h = hashlib.md5(f"{salt}|{j}|{g}".encode("utf-8")).hexdigest()
86:             pos = int(h, 16) % L
87:             bits[pos] = True
