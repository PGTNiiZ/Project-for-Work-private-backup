def find_root():
    candidates = [Path.cwd().resolve(), *Path.cwd().resolve().parents,

def source(module, *names):
    for name in names:
        display(Markdown(f'### `{module.__name__}.{name}`'))
        print(inspect.getsource(getattr(module, name)))
